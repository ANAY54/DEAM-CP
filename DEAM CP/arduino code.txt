// Function to get current time in RFC 3339 format for Firestore
String getFirestoreTimestamp() {
  struct tm timeinfo;

  if (!getLocalTime(&timeinfo, 5000)) {
    Serial.println("ERROR: Unable to obtain synchronized time.");
    return "";
  }

  int year = timeinfo.tm_year + 1900;

  if (year < 2024) {
    Serial.println("ERROR: Invalid system date.");
    return "";
  }

  time_t now;
  time(&now);

  struct tm utcTime;
  gmtime_r(&now, &utcTime);

  char timeString[25];

  strftime(
    timeString,
    sizeof(timeString),
    "%Y-%m-%dT%H:%M:%SZ",
    &utcTime
  );

  return String(timeString);
}
void printLocalTime() {
  Serial.println(getFirestoreTimestamp());
}

void loop() {
  static unsigned long lastPoll = 0;
  static unsigned long lastDebounceTime = 0;
  static unsigned long lastTimelapse = 0;
  unsigned long now = millis();

  // Physical doorbell button - press it just like a real Ring button.
  bool buttonReading = digitalRead(BUTTON_PIN);
  if (buttonReading != lastButtonState) {
    lastDebounceTime = now;
  }
  if (now - lastDebounceTime > BUTTON_DEBOUNCE_MS) {
    bool pressedEdge = (buttonReading == LOW && lastButtonState == HIGH);
    if (pressedEdge && !captureInProgress) {
      Serial.println("Doorbell button pressed");
      captureInProgress = true;
      xTaskCreate(
        captureAndUploadTask,
        "upload_task",
        8192,
        (void*) "button",
        1,
        NULL
      );
    }
  }
  lastButtonState = buttonReading;

  // Remote "Capture now" request from the web dashboard.
  if (!captureInProgress &&
      WiFi.status() == WL_CONNECTED &&
      now - lastPoll >= COMMAND_POLL_INTERVAL_MS) {
    lastPoll = now;

    if (checkRemoteCaptureTrigger()) {
      Serial.println("Remote capture requested from dashboard");
      captureInProgress = true;
      xTaskCreate(
        captureAndUploadTask,
        "upload_task",
        8192,
        (void*) "remote",
        1,
        NULL
      );
    }
  }

  // Periodic timelapse snapshot for the 24h timeline view.
  if (!captureInProgress &&
      WiFi.status() == WL_CONNECTED &&
      now - lastTimelapse >= TIMELAPSE_INTERVAL_MS) {
    lastTimelapse = now;
    captureInProgress = true;
    xTaskCreate(
      captureAndUploadTask,
      "upload_task",
      8192,
      (void*) "timelapse",
      1,
      NULL
    );
  }

  delay(50);
}

// Returns a valid Firebase ID token, reusing the cached one until it is
// close to expiring instead of signing in again on every poll.
String ensureIdToken() {
  unsigned long now = millis();
  bool expired = (cachedIdToken.length() == 0) ||
                 (now - idTokenFetchedAtMs > ID_TOKEN_TTL_MS);

  if (expired) {
    String token = getFirebaseIdToken();
    cachedIdToken = token;
    idTokenFetchedAtMs = now;
  }

  return cachedIdToken;
}

// Polls the "commands/smartdoorbell01" Firestore document the web dashboard
// writes to, and refreshes our heartbeat so the dashboard can show the
// device as online. Returns true (and clears the flag) if a capture was
// requested.
bool checkRemoteCaptureTrigger() {
  String idToken = ensureIdToken();
  if (idToken.length() == 0) {
    return false;
  }

  WiFiClientSecure commandClient;
  commandClient.setInsecure();

  HTTPClient http;
  if (!http.begin(commandClient, command_doc_url)) {
    return false;
  }
  http.addHeader("Authorization", "Bearer " + idToken);

  int httpCode = http.GET();

  if (httpCode == 401) {
    // Cached token was rejected (expired/revoked) - force a fresh sign-in.
    http.end();
    cachedIdToken = "";
    idToken = ensureIdToken();
    if (idToken.length() == 0) {
      return false;
    }
    if (!http.begin(commandClient, command_doc_url)) {
      return false;
    }
    http.addHeader("Authorization", "Bearer " + idToken);
    httpCode = http.GET();
  }

  if (httpCode != 200) {
    http.end();
    return false;
  }

  String response = http.getString();
  http.end();

  DynamicJsonDocument doc(512);
  if (deserializeJson(doc, response)) {
    return false;
  }

  bool shouldCapture =
      doc["fields"]["triggerCapture"]["booleanValue"] | false;

  sendHeartbeat(idToken, shouldCapture);

  return shouldCapture;
}

// Updates "lastSeen" so the dashboard can tell the device is online, and
// clears triggerCapture in the same request when a capture was just honored.
void sendHeartbeat(const String& idToken, bool resetTrigger) {
  String timestamp = getFirestoreTimestamp();
  if (timestamp.length() == 0) {
    return;
  }

  WiFiClientSecure heartbeatClient;
  heartbeatClient.setInsecure();

  HTTPClient http;
  const char* url =
      resetTrigger ? command_heartbeat_reset_url : command_heartbeat_url;

  if (!http.begin(heartbeatClient, url)) {
    return;
  }
  http.addHeader("Authorization", "Bearer " + idToken);
  http.addHeader("Content-Type", "application/json");

  DynamicJsonDocument doc(256);
  doc["fields"]["lastSeen"]["timestampValue"] = timestamp;
  if (resetTrigger) {
    doc["fields"]["triggerCapture"]["booleanValue"] = false;
  }

  String payload;
  serializeJson(doc, payload);

  http.sendRequest("PATCH", payload.c_str());
  http.end();
}

void captureAndUploadTask(void *parameter) {
  const char* trigger = (const char*) parameter;
  Serial.printf("📷 Starting capture task (trigger: %s)...\n", trigger);

  // Capture image
  fb = esp_camera_fb_get();
  if (!fb) {
    Serial.println("Camera capture failed");
    captureInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  Serial.printf("Image captured: %u bytes\n", fb->len);

  if (fb->len < 1000) {
    Serial.println("Image too small");
    esp_camera_fb_return(fb);
    captureInProgress = false;
    vTaskDelete(NULL);
    return;
  }

  // Upload to Cloudinary
  String imageUrl = uploadToCloudinary();

  // Free camera buffer
  esp_camera_fb_return(fb);
  fb = nullptr;

  if (imageUrl.length() > 0) {
    // Upload to Firestore
    uploadToFirestore(imageUrl, trigger);
  }

  Serial.println("Task completed");
  Serial.printf("Free heap after task: %d bytes\n", ESP.getFreeHeap());

  captureInProgress = false;

  // Delete this task
  vTaskDelete(NULL);
}

  client.println("Content-Length: " + String(contentLength));
  client.println();
 
  // Send header
  client.print(header);
 
  // Here we Send image data in small chunks
  const size_t chunkSize = 1024;
  for (size_t i = 0; i < fb->len; i += chunkSize) {
    size_t len = min(chunkSize, fb->len - i);
    client.write(fb->buf + i, len);
    vTaskDelay(1 / portTICK_PERIOD_MS); // Yield to other tasks
  }
 
  // Send footer
  client.print(footer);
 
  Serial.println("Data sent, waiting for response...");
 
  // Wait for the HTTP response

// Skip HTTP headers
// Read HTTP headers and detect response format
int responseContentLength = -1;
bool chunked = false;

while (client.connected() || client.available()) {
  String line = client.readStringUntil('\n');
  line.trim();

  if (line.length() == 0) {
    break;
  }

  if (line.startsWith("Content-Length:")) {
    responseContentLength = line.substring(15).toInt();
  }

  if (line.startsWith("Transfer-Encoding:") &&
      line.indexOf("chunked") >= 0) {
    chunked = true;
  }

  vTaskDelay(1 / portTICK_PERIOD_MS);
}

// Read the response body
cloudinaryResponse = "";

if (chunked) {
  while (client.connected() || client.available()) {
    String sizeLine = client.readStringUntil('\n');
    sizeLine.trim();

    int chunkSize = strtol(sizeLine.c_str(), NULL, 16);

    if (chunkSize <= 0) {
      // Consume the final empty line
      client.readStringUntil('\n');
      break;
    }

    int bytesRead = 0;
    unsigned long chunkStart = millis();

    while (bytesRead < chunkSize &&
           millis() - chunkStart < 15000) {

      while (client.available() && bytesRead < chunkSize) {
        cloudinaryResponse += (char)client.read();
        bytesRead++;
      }

      if (bytesRead < chunkSize) {
        vTaskDelay(1 / portTICK_PERIOD_MS);
      }
    }

    if (bytesRead != chunkSize) {
      Serial.println("Incomplete HTTP chunk");
      client.stop();
      return "";
    }

    // Consume CRLF after the chunk
    client.readStringUntil('\n');
  }

} else if (responseContentLength >= 0) {
  unsigned long start = millis();

  while (cloudinaryResponse.length() < responseContentLength &&
         millis() - start < 15000) {

    while (client.available() &&
           cloudinaryResponse.length() < responseContentLength) {
      cloudinaryResponse += (char)client.read();
    }

    vTaskDelay(1 / portTICK_PERIOD_MS);
  }
}

client.stop();

Serial.println("Cloudinary JSON response:");

if (error) {

}

if (!doc.containsKey("secure_url")) {
  Serial.println("No secure_url in response");
  return "";
}

String url = doc["secure_url"].as<String>();

Serial.println("Image uploaded: " + url);

String getFirebaseIdToken() {
  Serial.println("Signing in to Firebase...");

  WiFiClientSecure authClient;
  authClient.setInsecure();  // Testing only; see security note below

  HTTPClient http;

  String authUrl =
    "https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key="
    + String(firebase_api_key);

  if (!http.begin(authClient, authUrl)) {
    Serial.println("Could not start Firebase authentication request");
    return "";
  }

  http.addHeader("Content-Type", "application/json");

  DynamicJsonDocument requestDoc(512);

  requestDoc["email"] = firebase_email;
  requestDoc["password"] = firebase_password;
  requestDoc["returnSecureToken"] = true;

  String requestBody;
  serializeJson(requestDoc, requestBody);

  int httpCode = http.POST(requestBody);

  String response = http.getString();

  http.end();

  if (httpCode != 200) {
    Serial.printf("Firebase login failed: HTTP %d\n", httpCode);
    Serial.println(response);
    return "";
  }

  DynamicJsonDocument responseDoc(2048);

  DeserializationError error =
    deserializeJson(responseDoc, response);

  if (error) {
    Serial.println("Could not parse Firebase login response");
    return "";
  }

  if (responseDoc["idToken"].isNull()) {
    Serial.println("Firebase response did not contain an ID token");
    return "";
  }

  String idToken = responseDoc["idToken"].as<String>();

  Serial.println("Firebase authentication successful");

  return idToken;
}
